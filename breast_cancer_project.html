1. Data Collection

2. Data Pre-processing and Wrangling

We remove identifier columns, check for missing values, and convert the diagnosis into a factor for classification.

names(raw) <- trimws(names(raw))

if (!"diagnosis" %in% names(raw)) {
  stop("The dataset has no 'diagnosis' column. Check data.csv.")
}

df <- raw %>%
  select(-any_of(c("id", "X", "Unnamed: 32"))) %>%
  mutate(
    diagnosis = factor(
      diagnosis,
      levels=c("B", "M"),
      labels=c("Benign", "Malignant")
    )
  ) %>%
  filter(!is.na(diagnosis))

4. Model Training

We split the dataset into training and testing sets. The models learn from the training data and are evaluated on held-out test data.

idx <- createDataPartition(df$diagnosis, p=0.7, list=FALSE)
train <- df[idx, ]
test <- df[-idx, ]

tree <- rpart(diagnosis ~ ., data=train, method="class")

rf <- randomForest(
  diagnosis ~ ., data=train,
  ntree=500, importance=TRUE
)

cat("Training samples:", nrow(train), "\n")
## Training samples: 399
cat("Testing samples:", nrow(test), "\n")
## Testing samples: 170

rpart.plot(tree, extra=104,

5. Model Testing and Evaluation

We evaluate both models using a confusion matrix, accuracy, sensitivity, and specificity. These are educational model evaluation metrics, not clinical validation.

tree_pred <- predict(tree, test, type="class")
rf_pred <- predict(rf, test, type="class")

tc <- confusionMatrix(
  data=tree_pred, reference=test$diagnosis,
  positive="Malignant"
)

rc <- confusionMatrix(
  data=rf_pred, reference=test$diagnosis,
  positive="Malignant"
)

cat("Decision Tree confusion matrix:\n")
## Decision Tree confusion matrix:

5.1 ROC Curves

tree_prob <- predict(tree, test, type="prob")[, "Malignant"]
rf_prob <- predict(rf, test, type="prob")[, "Malignant"]

r1 <- roc(test$diagnosis, tree_prob,
          levels=c("Benign", "Malignant"), quiet=TRUE)
r2 <- roc(test$diagnosis, rf_prob,
          levels=c("Benign", "Malignant"), quiet=TRUE)

plot(r1, col="blue", lwd=2,
     main="ROC Curve Comparison")
plot(r2, col="red", lwd=2, add=TRUE)

legend("bottomright",
       legend=c(
         paste("Decision Tree AUC:", round(as.numeric(auc(r1)), 3)),
         paste("Random Forest AUC:", round(as.numeric(auc(r2)), 3))
       ),
       col=c("blue", "red"), lwd=2)
